\section{Osnova dizertačného projektu}

\begin{enumerate}
    \item Prehľad matematických nástrojov používaných pri metódach fúzie údajov zo snímačov.
    \item \textbf{Súhrn snímačov používaných pre lokalizáciu mobilných robotov a analýza princípov ich fungovania.} Globálny navigačný systém (GNSS), inerciálna jednotka (IMU), magnetometer, kolesová odometria, vizuálna odometria (monokamera, stereokamera), lidarová odometria, ultrazvukový senzor, radar.
    \item \textbf{Súhrn a analýza dostupných metód fúzie dát zo snímačov.} Bayesov filter, Kalmanov filter a jeho variácie, časticový filter a Monte Carlo.
    \item \textbf{Využitie metód umelej inteligencie a strojového učenia pri fúzii dát zo snímačov.} Umelé neurónové siete, rekurentné neurónové siete, učenie s posilňovaním.
    \item Príklady a porovnanie implementácií fúzie dát zo snímačov pre lokalizáciu mobilných robotov.
\end{enumerate}
